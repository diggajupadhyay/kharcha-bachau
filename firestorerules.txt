rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Users Collection
    // Users can only read/write their own user document
    match /users/{userId} {
      // Validate user document on creation — restrict to expected fields
      function isValidUserCreate() {
        return request.resource.data.keys().hasOnly(['id', 'name', 'email', 'customCategories', 'createdAt']) &&
               request.resource.data.id == userId &&
               request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 100 &&
               request.resource.data.email is string &&
               request.resource.data.customCategories is list &&
               request.resource.data.createdAt is number;
      }

      // Create: user must be authenticated and document must be valid
      allow create: if request.auth != null && request.auth.uid == userId && isValidUserCreate();
      
      // Read: user can only read their own document
      allow read: if request.auth != null && request.auth.uid == userId;
      
      // Update: only allow updating the customCategories field
      allow update: if request.auth != null && 
                      request.auth.uid == userId &&
                      request.resource.data.diff(resource.data).affectedKeys().hasOnly(['customCategories']) &&
                      request.resource.data.customCategories is list;
      
      // Delete: user can only delete their own document
      allow delete: if request.auth != null && request.auth.uid == userId;
    }

    // Wallets Collection
    // Allow access if the user's UID is in the 'members' array
    match /wallets/{walletId} {
      // Helper function to check if user is a member
      function isMember() {
        return request.auth != null && request.auth.uid in resource.data.members;
      }

      // Helper function to check if user is the owner
      function isOwner() {
        return request.auth != null && request.auth.uid == resource.data.ownerId;
      }

      // Validate new wallet data on creation
      function isValidWalletCreate() {
        return request.resource.data.keys().hasAll(['id', 'name', 'ownerId', 'members', 'currency', 'createdAt']) &&
               request.resource.data.id == walletId &&
               request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 50 &&
               request.resource.data.ownerId == request.auth.uid &&
               request.resource.data.members is list && request.resource.data.members.size() == 1 && request.resource.data.members[0] == request.auth.uid &&
               request.resource.data.currency is string && request.resource.data.currency.size() > 0 &&
               request.resource.data.createdAt is number;
      }

      // A shared wallet is not personal. Written with an explicit `in` check because
      // reading an absent field raises an error that denies the whole rule.
      function isShared() {
        return !('isPersonal' in resource.data && resource.data.isPersonal == true);
      }

      // Validate updated wallet data
      function isValidWalletUpdate() {
        let incomingData = request.resource.data;
        let existingData = resource.data;

        // Owner can update name and currency
        let ownerCanUpdate = incomingData.diff(existingData).affectedKeys().hasOnly(['name', 'currency']);
        // Any member may publish the wallet's invite code, but only once — making it
        // write-once stops one member silently rotating the code on everyone else.
        let memberCanPublishInviteCode = isMember() &&
                             incomingData.diff(existingData).affectedKeys().hasOnly(['inviteCode']) &&
                             !('inviteCode' in existingData) &&
                             incomingData.inviteCode is string &&
                             incomingData.inviteCode.size() >= 6 &&
                             incomingData.inviteCode.size() <= 12 &&
                             isShared();
        // The monthly budget is a shared planning figure, so any member may set it —
        // the Settings screen offers the field to everyone, not just the owner. Without
        // this branch every budget write was rejected and swallowed by a silent catch,
        // so budgets never synced and never reached a second device.
        let memberCanSetBudget = isMember() &&
                             incomingData.diff(existingData).affectedKeys().hasOnly(['budget']) &&
                             incomingData.budget is number &&
                             incomingData.budget >= 0 &&
                             incomingData.budget <= 1000000000000;
        // Each member may publish their own display name, and only their own. Nobody
        // gets to rename anyone else. Without this map the UI can only fall back to
        // "Member 8f3a", because users/{uid} is readable solely by its owner.
        let memberCanSetOwnProfile = isMember() &&
                             incomingData.diff(existingData).affectedKeys().hasOnly(['memberProfiles']) &&
                             incomingData.get('memberProfiles', {})
                               .diff(existingData.get('memberProfiles', {}))
                               .affectedKeys().hasOnly([request.auth.uid]) &&
                             incomingData.memberProfiles[request.auth.uid] is string &&
                             incomingData.memberProfiles[request.auth.uid].size() > 0 &&
                             incomingData.memberProfiles[request.auth.uid].size() <= 60;
        // Members can only update 'members' array (to leave). Exactly one entry may
        // disappear and it must be the caller's own: without the size and hasAll
        // checks a member could "leave" by submitting an empty list, dropping every
        // co-member — including the owner, who then could not even read the wallet.
        let memberCanLeave = incomingData.diff(existingData).affectedKeys().hasOnly(['members']) &&
                             incomingData.members is list &&
                             !(request.auth.uid in incomingData.members) && // User is removing themselves
                             (request.auth.uid in existingData.members) && // User was a member
                             incomingData.members.size() == existingData.members.size() - 1 &&
                             existingData.members.hasAll(incomingData.members); // nobody else dropped

        return (isOwner() && ownerCanUpdate) || (isMember() && memberCanLeave) ||
               memberCanPublishInviteCode || memberCanSetBudget || memberCanSetOwnProfile;
      }

      // Joining a shared wallet with an invite code. This branch is reachable by ANY
      // authenticated user who knows the wallet id, so it must be locked to exactly
      // one operation: append yourself to `members`, change nothing else. Without the
      // hasOnly/hasAll/size checks a "joiner" could rewrite ownerId and drop every
      // existing member, taking over the wallet and then deleting it.
      function isValidWalletJoin() {
        let incomingData = request.resource.data;
        let existingData = resource.data;

        return incomingData.diff(existingData).affectedKeys().hasOnly(['members']) &&
               incomingData.members is list &&
               existingData.members is list &&
               request.auth.uid in incomingData.members &&
               !(request.auth.uid in existingData.members) &&
               incomingData.members.hasAll(existingData.members) &&  // no member may be dropped
               incomingData.members.size() == existingData.members.size() + 1 &&
               incomingData.members.size() <= 50 &&
               isShared();
      }

      // Allow creation if authenticated, user is owner, and data is valid
      allow create: if request.auth != null && 
                       request.auth.uid in request.resource.data.members && 
                       isValidWalletCreate();
      
      // Allow read if user is a member
      allow read: if isMember();
      
      // Allow update if owner (name/currency), member leaving, member publishing the
      // invite code, or a new member joining a shared wallet.
      allow update: if request.auth != null && (
                       isValidWalletUpdate() ||
                       isValidWalletJoin()
                     );
      
      // Allow delete only if user is the wallet owner
      allow delete: if isOwner();

      // Subcollection: Expenses
      match /expenses/{expenseId} {
        // Helper function to get wallet data
        function getWallet() {
          return get(/databases/$(database)/documents/wallets/$(walletId)).data;
        }

        // Helper function to check if user is a member of the wallet
        function isWalletMember() {
          return request.auth != null && request.auth.uid in getWallet().members;
        }

        // Helper function to check if user is the wallet owner
        function isWalletOwner() {
          return request.auth != null && request.auth.uid == getWallet().ownerId;
        }

        // Helper function to check if user is the creator of the expense
        function isExpenseCreator() {
          return request.auth != null && request.auth.uid == resource.data.createdBy.uid;
        }

        // Validate new expense data on creation
        function isValidExpenseCreate() {
          // hasAll names what must be present; hasOnly caps what may be. Without the
          // second check a member could pad an expense with arbitrary fields up to
          // Firestore's 1 MiB document limit, and every other member of the wallet
          // would sync it down.
          let hasRequiredFields = request.resource.data.keys().hasAll(['id', 'walletId', 'categoryId', 'categoryName', 'categoryEmoji', 'amount', 'note', 'date', 'createdBy', 'createdAt']) &&
                 request.resource.data.keys().hasOnly(['id', 'walletId', 'categoryId', 'categoryName', 'categoryEmoji', 'amount', 'note', 'date', 'createdBy', 'createdAt', 'splitDetails', 'tags', 'transportDetails']) &&
                 request.resource.data.createdBy.keys().hasOnly(['uid', 'name']);
          let basicValidation = request.resource.data.id == expenseId &&
                 request.resource.data.walletId == walletId &&
                 request.resource.data.categoryId is string && request.resource.data.categoryId.size() > 0 &&
                 request.resource.data.categoryName is string && request.resource.data.categoryName.size() > 0 &&
                 request.resource.data.categoryEmoji is string && request.resource.data.categoryEmoji.size() > 0 &&
                  request.resource.data.amount is number && request.resource.data.amount > 0 && request.resource.data.amount <= 1000000000 &&
                 request.resource.data.note is string && request.resource.data.note.size() <= 500 &&
                 request.resource.data.date is string && request.resource.data.date.size() == 10 && // YYYY-MM-DD
                 request.resource.data.createdBy.uid == request.auth.uid &&
                 request.resource.data.createdBy.name is string && request.resource.data.createdBy.name.size() > 0 &&
                 request.resource.data.createdAt is number;
          
          // Validate splitDetails if present (must be valid if it exists, or can be absent/null)
          let splitDetailsValid = !('splitDetails' in request.resource.data) || 
            request.resource.data.splitDetails == null ||
            (request.resource.data.splitDetails.keys().hasAll(['splitType', 'participants', 'paidBy']) &&
             (request.resource.data.splitDetails.splitType == 'equal' || request.resource.data.splitDetails.splitType == 'percentage' || request.resource.data.splitDetails.splitType == 'custom') &&
             request.resource.data.splitDetails.participants is list && request.resource.data.splitDetails.participants.size() > 0 &&
             request.resource.data.splitDetails.paidBy is string && request.resource.data.splitDetails.paidBy.size() > 0 &&
             request.resource.data.splitDetails.paidBy in getWallet().members &&
             (!('settlements' in request.resource.data.splitDetails) || (request.resource.data.splitDetails.settlements is list)));
          
          // Validate tags if present (must be array of strings, max 5 tags)
          let tagsValid = !('tags' in request.resource.data) ||
            request.resource.data.tags == null ||
            (request.resource.data.tags is list && request.resource.data.tags.size() <= 5);
          
          // Validate transportDetails if present (for transport category)
          let transportDetailsValid = !('transportDetails' in request.resource.data) ||
            request.resource.data.transportDetails == null ||
            (request.resource.data.transportDetails.keys().hasAll(['passengers', 'from', 'to']) &&
             request.resource.data.transportDetails.passengers is number &&
             request.resource.data.transportDetails.passengers > 0 &&
             request.resource.data.transportDetails.from is string &&
             request.resource.data.transportDetails.from.size() > 0 &&
             request.resource.data.transportDetails.to is string &&
             request.resource.data.transportDetails.to.size() > 0);
          
          return hasRequiredFields && basicValidation && splitDetailsValid && tagsValid && transportDetailsValid;
        }

        // Validate updated expense data
        function isValidExpenseUpdate() {
          let incomingData = request.resource.data;
          let existingData = resource.data;

          // Only amount, note, splitDetails, tags, and transportDetails can be updated
          let allowedFields = ['amount', 'note', 'splitDetails', 'tags', 'transportDetails'];
          let affectedKeys = incomingData.diff(existingData).affectedKeys();

           let basicValidation = affectedKeys.hasOnly(allowedFields) &&
                  (incomingData.amount is number && incomingData.amount > 0 && incomingData.amount <= 1000000000) &&
                 // Same 500-character cap as create. Unbounded here, an edit could
                 // grow a note the create rule would have refused outright.
                 (incomingData.note is string && incomingData.note.size() <= 500);
          
          // Validate splitDetails if being updated (can be null to remove, or must be valid)
          let splitDetailsValid = !('splitDetails' in affectedKeys) ||
            (incomingData.splitDetails == null) ||
            (incomingData.splitDetails.keys().hasAll(['splitType', 'participants', 'paidBy']) &&
             (incomingData.splitDetails.splitType == 'equal' || incomingData.splitDetails.splitType == 'percentage' || incomingData.splitDetails.splitType == 'custom') &&
             incomingData.splitDetails.participants is list && incomingData.splitDetails.participants.size() > 0 &&
             incomingData.splitDetails.paidBy is string && incomingData.splitDetails.paidBy.size() > 0 &&
             incomingData.splitDetails.paidBy in getWallet().members &&
             (!('settlements' in incomingData.splitDetails) || (incomingData.splitDetails.settlements is list)));
          
          // Validate tags if being updated (can be null to remove, or must be valid array)
          let tagsValid = !('tags' in affectedKeys) ||
            (incomingData.tags == null) ||
            (incomingData.tags is list && incomingData.tags.size() <= 5);
          
          // Validate transportDetails if being updated (can be null to remove, or must be valid)
          let transportDetailsValid = !('transportDetails' in affectedKeys) ||
            (incomingData.transportDetails == null) ||
            (incomingData.transportDetails.keys().hasAll(['passengers', 'from', 'to']) &&
             incomingData.transportDetails.passengers is number &&
             incomingData.transportDetails.passengers > 0 &&
             incomingData.transportDetails.from is string &&
             incomingData.transportDetails.from.size() > 0 &&
             incomingData.transportDetails.to is string &&
             incomingData.transportDetails.to.size() > 0);
          
          return basicValidation && splitDetailsValid && tagsValid && transportDetailsValid;
        }

        // Allow create if user is a member and expense data is valid
        allow create: if isWalletMember() && isValidExpenseCreate();
        
        // Allow read if user is a member of the parent wallet
        allow read: if isWalletMember();
        
        // Validate that a single settlement entry is being appended and that the
        // caller is either the debtor or the payer (prevents one member from
        // marking someone else's debt as paid).
        function isValidSettlement() {
          let newSettlements = request.resource.data.splitDetails.get('settlements', []);
          // Read with a default: the client only creates `settlements` when the first
          // one is recorded, so on a brand-new expense the field is absent. Reading it
          // directly made `oldSettlements is list` false and denied the very first
          // settlement to anyone who was not the expense creator or the wallet owner —
          // i.e. to the person who actually owes the money.
          let oldSettlements = resource.data.splitDetails.get('settlements', []);
          let added = newSettlements[newSettlements.size() - 1];
          return newSettlements is list &&
                 oldSettlements is list &&
                 newSettlements.size() == oldSettlements.size() + 1 &&
                 // Strictly append-only. The size check alone left every earlier
                 // entry unpinned, so a member could inflate or forge a settlement
                 // recorded weeks ago in the same write that appends a valid one.
                 newSettlements == oldSettlements.concat([added]) &&
                 added.settledBy == request.auth.uid &&
                 (added.fromUserId == request.auth.uid || added.toUserId == request.auth.uid) &&
                 added.fromUserId in getWallet().members &&
                 added.toUserId in getWallet().members &&
                 added.amount is number && added.amount > 0;
        }

        // Reversal of a settlement (the Undo affordance shown after settling). Only
        // the person who recorded the settlement may take it back, and exactly one
        // entry may disappear per write — removeAll() yields the entries present in
        // the old list but not the new one, so it doubles as an "and nothing else
        // was altered" check when paired with the size comparison.
        function isValidSettlementReversal() {
          let newSettlements = request.resource.data.splitDetails.get('settlements', []);
          let oldSettlements = resource.data.splitDetails.get('settlements', []);
          let removed = oldSettlements.removeAll(newSettlements);
          return newSettlements is list &&
                 oldSettlements is list &&
                 newSettlements.size() == oldSettlements.size() - 1 &&
                 oldSettlements.hasAll(newSettlements) &&
                 removed.size() == 1 &&
                 removed[0].settledBy == request.auth.uid;
        }

        // A settlement write touches only splitDetails, and within splitDetails only
        // the settlements list. Pinning splitType/paidBy/participants matters: without
        // it a member could zero out their own share, or reassign paidBy to themselves
        // and flip who owes whom, in the very same write that records a settlement.
        function isSettlementShapedUpdate() {
          let incoming = request.resource.data.splitDetails;
          let existing = resource.data.splitDetails;
          return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['splitDetails']) &&
                 incoming != null &&
                 existing != null &&
                 incoming.keys().hasAll(['splitType', 'participants', 'paidBy']) &&
                 incoming.keys().hasOnly(['splitType', 'participants', 'paidBy', 'settlements']) &&
                 incoming.splitType == existing.splitType &&
                 incoming.paidBy == existing.paidBy &&
                 incoming.participants == existing.participants;
        }

        // Allow update if user is a member AND (created the expense OR is wallet owner) AND update is valid
        // OR if recording a settlement they are authorised for (their own debt or a
        // payment they received), OR reversing a settlement they themselves recorded.
        allow update: if isWalletMember() && (
                        ((isExpenseCreator() || isWalletOwner()) && isValidExpenseUpdate()) ||
                        (isSettlementShapedUpdate() && isValidSettlement()) ||
                        (isSettlementShapedUpdate() && isValidSettlementReversal())
                       );
        
        // Allow delete if user is a member AND (created the expense OR is wallet owner)
        allow delete: if isWalletMember() && 
                        (isExpenseCreator() || isWalletOwner());
      }
    }

    // Invites Collection
    // A code is a capability: knowing it grants join access, so possession is the
    // only check on `get`. Listing is NEVER allowed — `allow read` (get + list) would
    // let any signed-in user enumerate every code, and therefore every shared wallet,
    // in the database. Clients look a wallet's own code up via wallets/{id}.inviteCode.
    match /invites/{inviteCode} {
      allow get: if request.auth != null;
      allow list: if false;

      // Allow create if user is a member of the wallet being invited to
      // AND the wallet is not personal (personal wallets cannot be shared)
      allow create: if request.auth != null &&
                       get(/databases/$(database)/documents/wallets/$(request.resource.data.walletId)).data.members.hasAny([request.auth.uid]) &&
                       !('isPersonal' in get(/databases/$(database)/documents/wallets/$(request.resource.data.walletId)).data &&
                         get(/databases/$(database)/documents/wallets/$(request.resource.data.walletId)).data.isPersonal == true) &&
                       request.resource.data.keys().hasAll(['walletId', 'createdAt']) &&
                       request.resource.data.walletId is string && request.resource.data.walletId.size() > 0 &&
                       request.resource.data.createdAt is number;
      
      // Invites are immutable: an update could repoint a code that other people
      // already hold at a different wallet.
      allow update: if false;

      // Deletion is permitted only once the wallet the code points at is gone.
      // Wallet deletion used to strand its invite forever — the code stayed
      // resolvable, so anyone still holding it got "Wallet not found" instead of
      // "this invite is no longer valid", and the code could never be recycled.
      // A dangling invite grants access to nothing, so reaping one is harmless.
      allow delete: if request.auth != null &&
                       resource.data.walletId is string &&
                       !exists(/databases/$(database)/documents/wallets/$(resource.data.walletId));
    }
  }
}
